    actions = {
        Link("View Logs", onClick = { /* handle click */})
    }
)
        Parameters:
        text - The primary content of the banner, briefly describing the information it conveys.

        title - An optional title, rendered in bold, that appears above the text.
        icon - Slot for an optional icon displayed on the left of the text or title.
        actions - Slot for optional primary actions (usually links) to show at the bottom of the banner, below the text.
        actionIcons - Slot for secondary actions (usually icon buttons), such as closing the banner to show at the top right of the banner, to the right of the text or title.
        style - An InlineBannerStyle used to style the banner.
        textStyle - The base TextStyle used by the text and title.
